   507→export declare interface ParseParameters {

   522→    /**

   526→     * Default: `undefined`.
   527→     */
   528→    last?: number;
   529→    /**
   530→     * Collect per-page metadata such as embedded links, title, pageLabel, and dimensions;

   534→    parsePageInfo?: boolean;
   535→    /**

   538→     * Default: `false`.
   539→     */
   540→    parseHyperlinks?: boolean;
   541→    /**
   542→     * Enforce logical line breaks by inserting a newline when the vertical distance

   544→     * Useful to preserve paragraph/line structure when text items are emitted as separate segments.
   545→     * Default: `true`.
   546→     */
   547→    lineEnforce?: boolean;
   548→    /**

   550→     * Larger values make the parser more likely to start a new line between items.
   551→     * Default: `4.6`.
   552→     */
   553→    lineThreshold?: number;
   554→    /**
   555→     * String inserted between text items on the same line when a sufficiently large horizontal gap is detected.

   557→     * Default: `'\t'`.
   558→     */
   559→    cellSeparator?: string;
   560→    /**

   570→     * Default: `'\n-- page_number of total_number --'`.
   571→     */
   572→    pageJoiner?: string;
   573→    /**
   574→     * Optional string used to join text items when returning a page's text.

   576→     * Default: `undefined`.
   577→     */
   578→    itemJoiner?: string;
   579→    /**

   582→     * Useful for excluding tiny decorative or tracking images.
   583→     * Default: `80`.
   584→     * Disable: `0`.

   598→    /**

   602→    imageDataUrl?: boolean;
   603→    /**
   604→     * Applies to both getImage() and getScreenshot(): include the image as a binary buffer.

   606→     */
   607→    imageBuffer?: boolean;
   608→    /**

   610→     * Enables tags (MCID, role/props) and structural/accessibility information useful for mapping text ↔ structure.
   611→     * For plain text extraction it's usually false (trade-off: larger output).
   612→     * Default: `false`.
   613→     */
   614→    includeMarkedContent?: boolean;
   615→    /**
   616→     * When true, text normalization is NOT performed in the worker thread.
   617→     * For plain text extraction, normalizing in the worker (false) is usually recommended.
   618→     * Default: `false`.
   619→     */
   620→    disableNormalization?: boolean;
