     1→/**
     2→ * Convertisseur PowerPoint (`.pptx`) pour les CV structurés.
     3→ *
     4→ * Ce module transforme un objet {@link ParsedCv} (issu de l'extraction NVIDIA)
     5→ * en une présentation PowerPoint professionnelle à l'aide de la bibliothèque
     6→ * `pptxgenjs` v4.
     7→ *
     8→ * La présentation générée suit une mise en page moderne 16:9 (LAYOUT_WIDE,
     9→ * 13,33″ × 7,5″) avec une palette cohérente :
    10→ * - Couleur d'accent : vert émeraude `#10B981`
    11→ * - Texte principal : gris foncé `#1F2937`
    12→ * - Fond clair : `#F9FAFB`
    13→ * - Fond sombre (couverture, clôture) : `#1F2937`
    14→ *
    15→ * Structure des diapositives :
    16→ *  1. Couverture (nom, titre, contacts, score optionnel)
    17→ *  2. Profil / résumé (si `summary` est renseigné) + statistiques clés
    18→ *  3. Expérience professionnelle (1+ diapos, 3 expériences max par diapo)
    19→ *  4. Formation
    20→ *  5. Compétences & Langues (deux colonnes)
    21→ *  6. Projets & Certifications (si au moins une entrée existe)
    22→ *  7. Clôture « Merci » avec rappel des contacts
    23→ *
    24→ * Les libellés sont en français par défaut, et en anglais si
    25→ * `parsedCv.detectedLanguage` commence par `'en'`.
    26→ *
    27→ * @example
    28→ * ```ts
    29→ * const buffer = await generatePowerPointCv({ parsedCv, score: 85 })
    30→ * // → Buffer binaire .pptx prêt à être renvoyé par une route API
    31→ * ```
    32→ */

   204→/**
   205→ * Découpe un tableau en sous-tableaux de taille fixe.
   206→ * @param arr - Tableau source.
   207→ * @param size - Taille de chaque morceau.
   208→ * @returns Tableau de morceaux.
   209→ */
   210→function chunk<T>(arr: T[], size: number): T[][] {
   211→  if (size <= 0) return arr.length ? [arr] : []
   212→  const out: T[][] = []
   213→  for (let i = 0; i < arr.length; i += size) {
   214→    out.push(arr.slice(i, i + size))
   215→  }
   216→  return out
   217→}
   218→
   219→/**
   220→ * Tronque une chaîne à une longueur maximale, en ajoutant « … » si nécessaire.
   221→ * @param str - Chaîne source.
   222→ * @param max - Longueur maximale.
   223→ * @returns Chaîne éventuellement tronquée.
   224→ */
   225→function truncate(str: string, max: number): string {
   226→  if (!str) return ''
   227→  return str.length > max ? str.slice(0, max - 1).trimEnd() + '…' : str
   228→}

   230→/**
   231→ * Formate une plage de dates à partir d'une date de début et de fin.
   232→ * @param start - Date de début (texte libre).
   233→ * @param end - Date de fin (texte libre, ex. « présent »).
   234→ * @returns La chaîne formatée, ex. `2020 – présent`, ou chaîne vide.
   235→ */
   236→function formatDateRange(start?: string, end?: string): string {
   237→  const parts: string[] = []
   238→  if (start && start.trim()) parts.push(start.trim())
   239→  if (end && end.trim()) parts.push(end.trim())
   240→  if (parts.length === 0) return ''
   241→  return parts.join(' – ')
   242→}
   243→
   244→/**
   245→ * Découpe un bloc de description en lignes non vides, pour restituer
   246→ * les sauts de ligne éventuels extraits par le modèle.
   247→ * @param description - Texte brut potentiellement multi-lignes.
   248→ * @returns Tableau de lignes nettoyées.
   249→ */
   250→function splitDescriptionLines(description: string | undefined): string[] {
   251→  if (!description) return []
   252→  return description
   253→    .split(/\r?\n/)
   254→    .map((line) => line.replace(/^[\s•●▪\-*]+/, '').trim())
   255→    .filter((line) => line.length > 0)
   256→}

   258→/**
   259→ * Concatène les éléments de contact non vides avec le séparateur ` | `.
   260→ * @param parsedCv - CV structuré.
   261→ * @returns Ligne de contacts, ex. `jean@mail.com | 06 12 34 56 78 | Paris`.
   262→ */
   263→function buildContactLine(parsedCv: ParsedCv): string {
   264→  const p = parsedCv.personalInfo
   265→  const items: string[] = []
   266→  if (p.email) items.push(p.email)
   267→  if (p.phone) items.push(p.phone)
   268→  if (p.location) items.push(p.location)
   269→  if (p.linkedin) items.push(p.linkedin)
   270→  if (p.website) items.push(p.website)
   271→  if (p.github) items.push(p.github)
   272→  return items.join('  |  ')
   273→}

   275→/**
   276→ * Tente d'estimer le nombre d'années d'expérience à partir des dates
   277→ * extraites des expériences professionnelles.
   278→ *
   279→ * La fonction cherche des années à 4 chiffres (19xx / 20xx) dans les
   280→ * champs `startDate` et `endDate`. Si au moins deux années distinctes
   281→ * sont trouvées, la différence est retournée. Si une expérience se
   282→ * termine par « présent »/« current »/etc., l'année courante est utilisée
   283→ * comme borne supérieure.
   284→ *
   285→ * @param experiences - Liste des expériences professionnelles.
   286→ * @returns Nombre d'années estimé, ou `null` si non exploitable.
   287→ */
   288→function estimateYearsOfExperience(
   289→  experiences: WorkExperience[],
   290→): number | null {
   291→  const years: number[] = []
   292→  const currentYear = new Date().getFullYear()
   293→  for (const exp of experiences) {
   294→    for (const dateStr of [exp.startDate, exp.endDate]) {
   295→      if (!dateStr) continue
   296→      const matches = dateStr.match(/\b(19|20)\d{2}\b/g)
   297→      if (matches) {
   298→        for (const m of matches) years.push(parseInt(m, 10))
   299→      }
   300→    }
   301→  }
   302→  if (years.length < 2) return null
   303→  const min = Math.min(...years)
   304→  const max = Math.max(...years)
   305→  const hasPresent = experiences.some((e) =>
   306→    /présent|present|current|now|aujourd|hui/i.test(e.endDate ?? ''),
   307→  )
   308→  const effectiveMax = hasPresent ? Math.max(max, currentYear) : max
   309→  const diff = effectiveMax - min
   310→  return diff > 0 ? diff : null
   311→}

   313→/**
   314→ * Regroupe les compétences par catégorie, en plaçant les compétences sans
   315→ * catégorie sous une catégorie « Autres » (traduite).
   316→ * @param skills - Liste des compétences.
   317→ * @param otherLabel - Libellé localisé pour la catégorie « Autres ».
   318→ * @returns Map ordonnée `catégorie → compétences[]`. La catégorie « Autres »
   319→ *          est toujours en dernière position.
   320→ */
   321→function groupSkillsByCategory(
   322→  skills: Skill[],
   323→  otherLabel: string,
   324→): Map<string, Skill[]> {
   325→  const groups = new Map<string, Skill[]>()
   326→  for (const skill of skills) {
   327→    const category = (skill.category ?? '').trim() || otherLabel
   328→    if (!groups.has(category)) groups.set(category, [])
   329→    groups.get(category)!.push(skill)
   330→  }
   331→  // Réordonner pour placer « Autres » en dernier.
   332→  if (groups.has(otherLabel)) {
   333→    const other = groups.get(otherLabel)!
   334→    groups.delete(otherLabel)
   335→    groups.set(otherLabel, other)
   336→  }
   337→  return groups
   338→}

   382→      x: MARGIN_X + 0.25,

   403→/**

   407→ * @param pptx - Instance pptxgenjs.

   416→  score?: number,
   417→): void {
   418→  const slide = pptx.addSlide()
   419→  slide.background = { color: DARK_BG }
   420→
   421→  // Bande d'accent en haut.
   422→  slide.addShape('rect', {
   423→    x: 0,
   424→    y: 0,
   425→    w: PAGE_W,
   426→    h: 0.18,
   427→    fill: { color: ACCENT_COLOR },
   428→    line: { type: 'none' },
   429→  })
   430→
   431→  // Badge score en haut à droite.
   432→  if (typeof score === 'number' && Number.isFinite(score)) {
   433→    const clamped = Math.max(0, Math.min(100, Math.round(score)))
   434→    slide.addText(`${labels.score} : ${clamped}/100`, {
   435→      x: PAGE_W - 4.5,
   436→      y: 0.45,
   437→      w: 3.9,
   438→      h: 0.5,
   439→      fontSize: FONT_SIZE_META,
   440→      bold: true,
   441→      color: DARK_BG,
   442→      align: 'right',
   443→      valign: 'middle',
   444→      fill: { color: ACCENT_COLOR },
   445→      rectRadius: 0.1,
   446→      shape: 'roundRect',
   447→    })
   448→  }
   449→
   450→  const p = parsedCv.personalInfo
   451→  const fullName = (p.fullName ?? '').trim()
   452→
   453→  // Nom complet centré.
   454→  slide.addText(fullName || '—', {
   455→    x: 0.5,
   456→    y: 2.3,
   457→    w: PAGE_W - 1,
   458→    h: 1.4,
   459→    fontSize: FONT_SIZE_COVER_NAME,
   460→    bold: true,
   461→    color: WHITE,
   462→    align: 'center',
   463→    valign: 'middle',
   464→    fontFace: 'Calibri',

   466→
   467→  // Titre professionnel sous le nom.
   468→  if (p.title) {

   470→      x: 0.5,
   471→      y: 3.75,
   472→      w: PAGE_W - 1,
   473→      h: 0.7,
   474→      fontSize: FONT_SIZE_COVER_TITLE,
   475→      color: ACCENT_LIGHT,
   476→      italic: true,
   477→      align: 'center',
   478→      valign: 'middle',
   479→      fontFace: 'Calibri',
   480→    })
   481→  }
   482→
   483→  // Ligne de contacts en bas.

   487→      x: PAGE_W / 2 - 3,
   488→      y: 6.0,
   489→      w: 6,
   490→      h: 0,
   491→      line: { color: ACCENT_COLOR, width: 1 },
   492→    })
   493→    slide.addText(contactLine, {
   494→      x: 0.5,
   495→      y: 6.15,
   496→      w: PAGE_W - 1,
   497→      h: 0.6,
   498→      fontSize: FONT_SIZE_CONTACT,
   499→      color: WHITE_MUTED,
   500→      align: 'center',
   501→      valign: 'middle',

   546→/**
   547→ * Ajoute une barre de statistiques clés (années d'expérience, compétences,
   548→ * langues) sur une diapositive. Les statistiques vides sont omises.
   549→ *
   550→ * @param slide - Diapositive cible.
   551→ * @param parsedCv - CV structuré.
   552→ * @param labels - Libellés localisés.
   553→ * @param y - Position verticale de la barre.
   554→ */
   555→function addStatsBar(
   556→  slide: pptxgen.Slide,
   557→  parsedCv: ParsedCv,
   558→  labels: CvLabels,
   559→  y: number,
   560→): void {
   561→  const stats: Array<{ value: string; label: string }> = []
   562→
   563→  const years = estimateYearsOfExperience(parsedCv.workExperience)
   564→  if (years !== null && years > 0) {
   565→    stats.push({ value: String(years), label: labels.yearsOfExperience })
   566→  } else if (parsedCv.workExperience.length > 0) {
   567→    stats.push({
   568→      value: String(parsedCv.workExperience.length),
   569→      label: labels.experiencesCount,
   570→    })
   571→  }

   628→/**

  1231→/**
  1232→ * Génère une présentation PowerPoint professionnelle à partir d'un CV
  1233→ * structuré.
  1234→ *
  1235→ * Le fichier généré suit une mise en page 16:9 moderne. Les diapositives
  1236→ * vides (sections sans données) sont automatiquement omises.
  1237→ *
  1238→ * @param params - Paramètres de génération.
  1239→ * @param params.parsedCv - CV structuré issu de l'extraction NVIDIA.
  1240→ * @param params.score - Score optionnel sur 100 (affiché sur la couverture).
  1241→ * @returns Un `Buffer` Node.js contenant le fichier `.pptx` binaire.
  1242→ *
  1243→ * @throws {Error} Si la génération échoue ou si le type de sortie renvoyé
  1244→ *                 par `pptxgenjs` est inattendu.
  1245→ */
  1246→export async function generatePowerPointCv(params: {
  1247→  parsedCv: ParsedCv
  1248→  score?: number
  1249→}): Promise<Buffer> {
  1250→  const { parsedCv, score } = params
  1251→
  1252→  const pptx = new pptxgen()
  1253→  // Layout 16:9 large (13,33″ × 7,5″) — équivalent PowerPoint standard.
  1254→  pptx.layout = 'LAYOUT_WIDE'
  1255→  pptx.author = 'Agent de transformation de CV'
  1256→  pptx.company = 'CV Transformer'
  1257→  pptx.subject = 'CV généré automatiquement'
  1258→  const fullName = (parsedCv.personalInfo.fullName ?? '').trim()
  1259→  pptx.title = fullName ? `CV — ${fullName}` : 'CV'
  1260→
  1261→  const labels = getLabels(parsedCv)
  1262→
  1263→  // Diapo 1 — Couverture.
  1264→  addCoverSlide(pptx, parsedCv, labels, score)
  1265→
  1266→  // Diapo 2 — Profil (si résumé présent).
  1267→  addProfileSlide(pptx, parsedCv, labels)
  1268→
  1269→  // Diapos 3+ — Expérience professionnelle (1+ diapos, 3 max par diapo).
  1270→  if (parsedCv.workExperience.length > 0) {
  1271→    const chunks = chunk(parsedCv.workExperience, MAX_EXPERIENCES_PER_SLIDE)
  1272→    chunks.forEach((expChunk, i) => {
  1273→      addExperienceSlide(pptx, expChunk, labels, i + 1, chunks.length)
  1274→    })
  1275→  }
  1276→
  1277→  // Diapo — Formation.
  1278→  addEducationSlide(pptx, parsedCv.education, labels)
  1279→
  1280→  // Diapo — Compétences & Langues.
  1281→  addSkillsAndLanguagesSlide(pptx, parsedCv, labels)
  1282→
  1283→  // Diapo — Projets & Certifications.
  1284→  addProjectsAndCertificationsSlide(pptx, parsedCv, labels)
  1285→
  1286→  // Diapo finale — Merci.
  1287→  addThankYouSlide(pptx, parsedCv, labels)
  1288→
  1289→  // Génération du buffer binaire.
  1290→  // En Node, `pptxgenjs` retourne un `Buffer` pour `nodebuffer`, mais le
  1291→  // type déclaré est une union plus large — on normalise donc défensivement.
  1292→  const result = await pptx.write({ outputType: 'nodebuffer' })
  1293→
  1294→  if (typeof result === 'string') {
  1295→    throw new Error(
  1296→      "Type de sortie inattendu pour la présentation PowerPoint : string au lieu d'un buffer.",
  1297→    )
  1298→  }
  1299→  if (Buffer.isBuffer(result)) {
  1300→    return result
  1301→  }
  1302→  if (result instanceof Uint8Array) {
  1303→    return Buffer.from(result)
  1304→  }
  1305→  if (result instanceof ArrayBuffer) {
  1306→    return Buffer.from(result)
  1307→  }
  1308→  // Cas restant (Blob) — non géré côté serveur Node.
  1309→  throw new Error(
  1310→    "Type de sortie inattendu pour la présentation PowerPoint (Blob non supporté).",
  1311→  )
  1312→}
  1313→
  1314→/**
  1315→ * Génère le nom de fichier suggéré pour la présentation PowerPoint.
  1316→ *
  1317→ * Le nom est normalisé : minuscules, diacritiques supprimés, caractères non
  1318→ * alphanumériques remplacés par `_`, préfixe `CV_`, extension `.pptx`.
  1319→ *
  1320→ * @param fullName - Nom complet du candidat.
  1321→ * @returns Nom de fichier sécurisé, ex. `CV_jean_dupont.pptx`.
  1322→ *
  1323→ * @example
  1324→ * ```ts
  1325→ * getPowerPointFileName('Jean Dupont')      // → 'CV_jean_dupont.pptx'
  1326→ * getPowerPointFileName("Marie-Josée O'Neil") // → 'CV_marie_josee_o_neil.pptx'
  1327→ * getPowerPointFileName('')                   // → 'CV_candidat.pptx'
  1328→ * ```
  1329→ */
  1330→export function getPowerPointFileName(fullName: string): string {
  1331→  const normalized = (fullName ?? '')
  1332→    .normalize('NFD')
  1333→    .replace(/[\u0300-\u036f]/g, '') // suppression des diacritiques
  1334→    .replace(/[^a-zA-Z0-9]+/g, '_') // tout caractère non alphanumérique → _
  1335→    .replace(/^_+|_+$/g, '') // trimming des underscores
  1336→    .toLowerCase()
  1337→  const base = normalized || 'candidat'
  1338→  return `CV_${base}.pptx`
  1339→}
